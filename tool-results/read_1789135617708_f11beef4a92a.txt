    52→function toMinutes(time: string): number {
    53→  const m = time.match(/(\d+):(\d+)\s*(AM|PM)/i);
    54→  if (!m) return 0;
    55→  let h = Number(m[1]) % 12;
    56→  if (/pm/i.test(m[3])) h += 12;
    57→  return h * 60 + Number(m[2]);
    58→}

    60→/** The student's / everyone's next upcoming class from the weekly routine. */
    61→function findNextClass(now: Date): { row: RoutineClass; label: string } | null {
    62→  const todayName = new Intl.DateTimeFormat("en-US", { weekday: "long" }).format(now);
    63→  const todayIdx = Math.max(0, weekDays.indexOf(todayName as WeekDay));
    64→  const nowMinutes = now.getHours() * 60 + now.getMinutes();
    65→
    66→  for (let offset = 0; offset < 8; offset++) {
    67→    const day = weekDays[(todayIdx + offset) % 7];
    68→    const rows = classRoutine
    69→      .filter((r) => r.day === day)
    70→      .sort((a, b) => toMinutes(a.start) - toMinutes(b.start));
    71→    for (const row of rows) {
    72→      if (offset === 0 && toMinutes(row.start) <= nowMinutes + 10) continue;
    73→      const label = offset === 0 ? "Today" : offset === 1 ? "Tomorrow" : day;
    74→      return { row, label };
    75→    }
    76→  }
    77→  return null;
    78→}

   332→function PortalRoutine({ courseSlug }: { courseSlug: string }) {
   333→  const today = useToday();
   334→  const [day, setDay] = useState<WeekDay | null>(null);
   335→  const [mineOnly, setMineOnly] = useState(true);
   336→
   337→  const activeDay: WeekDay =
   338→    day ?? ((weekDays as readonly string[]).includes(today) ? (today as WeekDay) : "Saturday");
   339→
   340→  const isMine = (row: RoutineClass) => row.courseSlug === courseSlug || isShared(row);
   341→  const allRows = classRoutine.filter((r) => r.day === activeDay);
   342→  const visibleRows = mineOnly ? allRows.filter(isMine) : allRows;
   343→
   344→  return (
   345→    <Reveal delay={0.05}>
   346→      <div className="rounded-3xl border border-border bg-card p-5 md:p-6">

   349→          <div className="flex rounded-full border border-border bg-[#101014] p-1">
   350→            {(["mine", "all"] as const).map((mode) => {
   351→              const active = mineOnly === (mode === "mine");

   353→                <button
   354→                  key={mode}

   356→                  onClick={() => setMineOnly(mode === "mine")}
   357→                  aria-pressed={active}
   358→                  className={`rounded-full px-3.5 py-1.5 text-xs font-semibold transition-all ${

   360→                      ? "bg-gold-gradient text-[#16120a]"
