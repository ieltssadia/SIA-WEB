   156→function AccountStep({ onAuthed }: { onAuthed: () => void }) {
   157→  const setSession = usePortalStore((s) => s.setSession);
   158→
   159→  const [mode, setMode] = useState<"signup" | "login">("signup");
   160→  const [name, setName] = useState("");
   161→  const [phone, setPhone] = useState("");
   162→  const [password, setPassword] = useState("");
   163→  const [confirm, setConfirm] = useState("");
   164→  const [error, setError] = useState<string | null>(null);
   165→  const [busy, setBusy] = useState(false);
   166→
   167→  async function submit(e: FormEvent<HTMLFormElement>) {
   168→    e.preventDefault();
   169→    setError(null);
   170→    if (mode === "signup" && password !== confirm) {
   171→      setError("Passwords don't match — আবার লিখুন।");
   172→      return;
   173→    }
   174→    setBusy(true);
   175→    try {
   176→      const res = await fetch(
   177→        mode === "signup" ? "/api/auth/register" : "/api/portal/login",
   178→        {
   179→          method: "POST",
   180→          headers: { "Content-Type": "application/json" },
   181→          body: JSON.stringify(
   182→            mode === "signup" ? { name, phone, password } : { phone, password }
   183→          ),
   184→        }
   185→      );
   186→      const data = (await res.json().catch(() => null)) as
   187→        | { user?: { name: string; phone: string }; token?: string; error?: string }
   188→        | null;
   189→      if (!res.ok || !data?.user) {
   190→        setError(data?.error ?? "Something went wrong — please try again.");
   191→        return;
   192→      }
   193→      setSession(data.user, [], [], data.token ?? null);
   194→      onAuthed();
   195→    } catch {
   196→      setError("Could not reach the server — check your connection.");
   197→    } finally {
   198→      setBusy(false);
   199→    }
   200→  }
   201→
   202→  return (
   203→    <div>
   204→      <h2 className="font-display text-lg font-bold text-foreground">
   205→        {mode === "signup" ? "Create your student account" : "Log in to continue"}
   206→      </h2>
   207→      <p className="mt-1 text-sm text-muted-foreground">
   208→        কোর্স কিনলে একই account-এ ভর্তি হবে — পোর্টালে সাথে সাথে দেখতে পাবেন।
   209→      </p>
   210→
   211→      <Tabs value={mode} onValueChange={(v) => { setMode(v as "signup" | "login"); setError(null); }} className="mt-4">
   212→        <TabsList className="grid w-full grid-cols-2 bg-[#101014]">
   213→          <TabsTrigger value="signup" className="gap-1.5 data-[state=active]:bg-gold-gradient data-[state=active]:text-[#16120a]">
   214→            <UserPlus className="h-4 w-4" aria-hidden /> Create account
   215→          </TabsTrigger>
   216→          <TabsTrigger value="login" className="gap-1.5 data-[state=active]:bg-gold-gradient data-[state=active]:text-[#16120a]">
   217→            <LogIn className="h-4 w-4" aria-hidden /> Log in
   218→          </TabsTrigger>
   219→        </TabsList>
   220→
   221→        <TabsContent value="signup">
   222→          <form onSubmit={submit} className="mt-4 space-y-3.5" noValidate>
   223→            <div className="space-y-1.5">
   224→              <Label htmlFor="co-name">Full name</Label>
   225→              <Input
   226→                id="co-name"
   227→                placeholder="e.g. Rahim Ahmed"
   228→                value={name}
   229→                onChange={(e) => { setName(e.target.value); setError(null); }}
   230→                autoComplete="name"
   231→              />
   232→            </div>
   233→            <PhoneField phone={phone} setPhone={setPhone} setError={setError} />
   234→            <div className="grid gap-3.5 sm:grid-cols-2">
   235→              <div className="space-y-1.5">
   236→                <Label htmlFor="co-pass">Password (min 6)</Label>
   237→                <Input
   238→                  id="co-pass"
   239→                  type="password"
   240→                  placeholder="Create a password"
   241→                  value={password}
   242→                  onChange={(e) => { setPassword(e.target.value); setError(null); }}
   243→                  autoComplete="new-password"
   244→                />
   245→              </div>
   246→              <div className="space-y-1.5">
   247→                <Label htmlFor="co-pass2">Confirm password</Label>
   248→                <Input
   249→                  id="co-pass2"
   250→                  type="password"
   251→                  placeholder="Repeat password"
   252→                  value={confirm}
   253→                  onChange={(e) => { setConfirm(e.target.value); setError(null); }}
   254→                  autoComplete="new-password"
   255→                />
   256→              </div>
   257→            </div>
   258→            {error ? <ErrorNote text={error} /> : null}
   259→            <Button type="submit" disabled={busy} className="w-full bg-gold-gradient font-semibold text-[#16120a] hover:opacity-90 disabled:opacity-60">
   260→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden /> : null}
   261→              Continue to payment
   262→              <ArrowRight className="ml-1.5 h-4 w-4" aria-hidden />
   263→            </Button>
   264→          </form>
   265→        </TabsContent>
   266→
   267→        <TabsContent value="login">
   268→          <form onSubmit={submit} className="mt-4 space-y-3.5" noValidate>
   269→            <PhoneField phone={phone} setPhone={setPhone} setError={setError} />
   270→            <div className="space-y-1.5">
   271→              <Label htmlFor="co-login-pass">Password</Label>
   272→              <Input
   273→                id="co-login-pass"
   274→                type="password"
   275→                placeholder="Your portal password"
   276→                value={password}
   277→                onChange={(e) => { setPassword(e.target.value); setError(null); }}
   278→                autoComplete="current-password"
   279→              />
   280→            </div>
   281→            {error ? <ErrorNote text={error} /> : null}
   282→            <Button type="submit" disabled={busy} className="w-full bg-gold-gradient font-semibold text-[#16120a] hover:opacity-90 disabled:opacity-60">

   286→            </Button>
   287→          </form>
   288→        </TabsContent>
   289→      </Tabs>
   290→    </div>
   291→  );

   338→
   339→function PaymentStep({
   340→  course,
   341→  onEnrolled,
   342→  onBack,
   343→}: {
   344→  course: Course;
   345→  /** Called with the confirmed batch label once enrollment succeeds. */
   346→  onEnrolled: (batch: string) => void;
   347→  onBack: () => void;
   348→}) {
   349→  const token = usePortalStore((s) => s.token);
   350→  const setSession = usePortalStore((s) => s.setSession);
   351→  const logout = usePortalStore((s) => s.logout);
   352→  const batchOptions = useMemo(() => nextBatchOptions(course.slug), [course.slug]);
   353→
   354→  const [batch, setBatch] = useState(batchOptions[0].batch);
   355→  const [method, setMethod] = useState<PayMethod["id"]>("bKash");
   356→  const [txId, setTxId] = useState("");
   357→  const [error, setError] = useState<string | null>(null);
   358→  const [busy, setBusy] = useState(false);
   359→
   360→  const selected = payMethods.find((m) => m.id === method)!;
   361→
   362→  // Reset batch if options change (course switched)
   363→  const [prevCourse, setPrevCourse] = useState(course.slug);
   364→  if (prevCourse !== course.slug) {
   365→    setPrevCourse(course.slug);
   366→    setBatch(batchOptions[0].batch);
   367→  }
   368→
   369→  async function confirm() {
   370→    setError(null);
   371→    setBusy(true);
   372→    try {
   373→      const res = await fetch("/api/portal/enroll", {
   374→        method: "POST",
   375→        headers: {
   376→          "Content-Type": "application/json",
   377→          Authorization: `Bearer ${token ?? ""}`,
   378→        },
   379→        body: JSON.stringify({
   380→          courseSlug: course.slug,
   381→          batch,
   382→          paymentMethod: method,
   383→          transactionId: txId,
   384→        }),
   385→      });
   386→      const data = (await res.json().catch(() => null)) as
   387→        | {
   388→            user?: { name: string; phone: string };
   389→            enrollments?: never[];
   390→            mocks?: never[];
   391→            token?: string;
   392→            error?: string;
   393→          }
   394→        | null;
   395→      if (res.status === 401) {
   396→        logout();
   397→        setError("Session expired — please log in again.");
   398→        onBack();
   399→        return;
   400→      }
   401→      if (!res.ok || !data?.user) {
   402→        setError(data?.error ?? "Enrollment failed — please try again.");
   403→        return;
   404→      }
   405→      setSession(data.user, data.enrollments ?? [], data.mocks ?? [], data.token ?? null);
   406→      onEnrolled(batch);
   407→    } catch {
   408→      setError("Could not reach the server — check your connection.");
   409→    } finally {
   410→      setBusy(false);
   411→    }
   412→  }
   413→
   414→  return (
   415→    <div className="space-y-6">
   416→      <div>
   417→        <h2 className="font-display text-lg font-bold text-foreground">Choose your batch</h2>
   418→        <RadioGroup value={batch} onValueChange={setBatch} className="mt-3 space-y-2">
   419→          {batchOptions.map((b) => (
   420→            <Label
   421→              key={b.batch}
   422→              htmlFor={`batch-${b.batch}`}
   423→              className={`flex cursor-pointer items-center gap-3 rounded-xl border p-3.5 transition-colors ${
   424→                batch === b.batch ? "border-primary/60 bg-primary/[0.07]" : "border-border bg-[#101014] hover:border-primary/30"
   425→              }`}
   426→            >
   427→              <RadioGroupItem id={`batch-${b.batch}`} value={b.batch} />
   428→              <span className="min-w-0">
   429→                <span className="block text-sm font-bold text-foreground">{b.batch}</span>
   430→                <span className="block text-xs text-muted-foreground">{b.meta}</span>
   431→              </span>
   432→              <CalendarDays className="ml-auto h-4 w-4 shrink-0 text-primary" aria-hidden />
   433→            </Label>
   434→          ))}
   435→        </RadioGroup>
   436→      </div>
   437→
   438→      <Separator className="bg-primary/10" />
   439→
   440→      <div>
   441→        <h2 className="font-display text-lg font-bold text-foreground">Payment method</h2>
   442→        <RadioGroup value={method} onValueChange={(v) => setMethod(v as PayMethod["id"])} className="mt-3 space-y-2">
   443→          {payMethods.map((m) => {
   444→            const Icon = m.icon;
   445→            return (
   446→              <Label
   447→                key={m.id}
   448→                htmlFor={`pay-${m.id}`}
   449→                className={`flex cursor-pointer items-center gap-3 rounded-xl border p-3.5 transition-colors ${
   450→                  method === m.id ? "border-primary/60 bg-primary/[0.07]" : "border-border bg-[#101014] hover:border-primary/30"
   451→                }`}
   452→              >
   453→                <RadioGroupItem id={`pay-${m.id}`} value={m.id} />
   454→                <span
   455→                  className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg text-white"
   456→                  style={{ backgroundColor: m.color }}
   457→                  aria-hidden
   458→                >
   459→                  <Icon className="h-4.5 w-4.5" />
   460→                </span>
   461→                <span className="min-w-0">
   462→                  <span className="block text-sm font-bold text-foreground">{m.label}</span>
   463→                  <span className="block truncate text-xs text-muted-foreground">{m.note}</span>
   464→                </span>
   465→              </Label>
   466→            );
   467→          })}
   468→        </RadioGroup>
   469→
   470→        {selected.needsTxId ? (
   471→          <div className="mt-3 space-y-1.5">
   472→            <Label htmlFor="co-txid">Transaction ID (optional here)</Label>
   473→            <Input
   474→              id="co-txid"
   475→              placeholder="Demo checkout — any text works"
   476→              value={txId}
   477→              onChange={(e) => setTxId(e.target.value)}
   478→              maxLength={60}
   479→            />
   480→            <p className="text-xs text-muted-foreground">
   481→              ডেমো মোড — আসল পেমেন্ট হয় না। প্রোডাকশনে TrxID ভেরিফিকেশন যুক্ত হবে।
   482→            </p>
   483→          </div>
   484→        ) : null}
   485→      </div>
   486→
   487→      {error ? <ErrorNote text={error} /> : null}
   488→
   489→      <div className="flex flex-col gap-2.5 sm:flex-row">
   490→        <Button
   491→          onClick={confirm}
   492→          disabled={busy}
   493→          className="flex-1 bg-gold-gradient py-6 text-base font-semibold text-[#16120a] shadow-[0_8px_30px_rgba(212,175,55,0.25)] hover:opacity-90 disabled:opacity-60"
   494→        >
   495→          {busy ? (
   496→            <>
   497→              <Loader2 className="mr-2 h-5 w-5 animate-spin" aria-hidden />
   498→              Processing...
   499→            </>
   500→          ) : (
   501→            <>
   502→              <ShieldCheck className="mr-2 h-5 w-5" aria-hidden />
   503→              Confirm Enrollment · {course.price === 0 ? "Free" : course.price ? taka(course.price) : "Custom"}
   504→            </>
   505→          )}
   506→        </Button>
   507→        <Button variant="outline" onClick={onBack} disabled={busy} className="border-primary/25 hover:text-primary">
   508→          Back
   509→        </Button>
   510→      </div>
   511→
   512→      <p className="flex items-center justify-center gap-1.5 text-xs text-muted-foreground">
   513→        <Lock className="h-3.5 w-3.5" aria-hidden />
   514→        Secure demo checkout · bKash / Nagad / Rocket / Bank / Cash accepted
   515→      </p>
   516→    </div>
   517→  );
   518→}

   614→
   615→export function CheckoutPage({ initialCourse }: { initialCourse: string | null }) {
   616→  const user = usePortalStore((s) => s.user);
   617→  const enrollments = usePortalStore((s) => s.enrollments);
   618→  const hasHydrated = usePortalStore((s) => s.hasHydrated);
   619→  const cartItems = useCartStore((s) => s.items);
   620→  const cartLastOrder = useCartStore((s) => s.lastOrder);
   621→  const cartHydrated = useCartStore((s) => s.hasHydrated);
   622→
   623→  const [courseSlug, setCourseSlug] = useState(initialCourse);
   624→  const [step, setStep] = useState<1 | 2 | 3>(1);
   625→  const [enrolledBatch, setEnrolledBatch] = useState<string | null>(null);
   626→
   627→  // Logged-in visitors skip account creation (render-time state adjustment).
   628→  // prevAuthed starts false so a visitor who arrives while ALREADY logged in
   629→  // still advances from step 1 → 2 on the first post-hydration render.
   630→  const authed = hasHydrated && !!user;
   631→  const [prevAuthed, setPrevAuthed] = useState(false);
   632→  if (authed !== prevAuthed) {
   633→    setPrevAuthed(authed);
   634→    if (authed) setStep((s) => (s === 1 ? 2 : s));
   635→    else setStep(1);
   636→  }
   637→
   638→  // The course comes from the hash query — sync prop → state when the
   639→  // visitor switches courses (render-time adjustment; useState ignores
   640→  // new prop values on re-render)
   641→  const [prevInitial, setPrevInitial] = useState(initialCourse);
   642→  if (initialCourse !== prevInitial) {
   643→    setPrevInitial(initialCourse);
   644→    setCourseSlug(initialCourse);
   645→    setEnrolledBatch(null);
   646→    setStep(authed ? 2 : 1);
   647→  }
   648→
   649→  const course = courses.find((c) => c.slug === courseSlug) ?? null;
   650→  const alreadyEnrolled = !!(course && user && enrollments.some((e) => e.courseSlug === course.slug));
   651→
   652→  /* --- No course chosen → cart checkout (+ course picker when cart is empty) --- */
   653→  if (!course) {
   654→    const cartHasItems = cartHydrated && cartItems.length > 0;
   655→    const showReceipt = cartHydrated && !cartHasItems && !!cartLastOrder;
   656→    return (
   657→      <>
   658→        <PageHeader
   659→          eyebrow={cartHasItems || showReceipt ? "Book Shop · Checkout" : "Admission · Checkout"}
   660→          title={
   661→            cartHasItems ? (
   662→              <>
   663→                Review &amp; <span className="text-gold-gradient">place your order</span>
   664→              </>
   665→            ) : showReceipt ? (
   666→              <>
   667→                Your recent <span className="text-gold-gradient">order</span>
   668→              </>
   669→            ) : (
   670→              <>
   671→                Choose a <span className="text-gold-gradient">course</span>
   672→              </>
   673→            )
   674→          }
   675→          subtitle={
   676→            cartHasItems
   677→              ? "ডেলিভারি ডিটেইলস দিন, পেমেন্ট বেছে নিন — অর্ডার কনফার্ম করতে ২৪ ঘণ্টার মধ্যে আমরা কল দেব।"
   678→              : showReceipt
   679→                ? "আপনার সর্বশেষ অর্ডারের রসিদ — পেমেন্ট ও ডেলিভারি তথ্য একসাথে।"
   680→                : "যে কোর্সে ভর্তি হতে চান বেছে নিন — payment-এর পরই কোর্সটি আপনার Student Portal-এ চলে আসবে।"
   681→          }
   682→        />
   683→        <section className="py-10 md:py-14">
   684→          <div className="mx-auto max-w-7xl px-4 lg:px-8">
   685→            <CartCheckout />
   686→
   687→            {/* Course picker stays reachable when the cart is empty */}
   688→            {cartHydrated && !cartHasItems && !cartLastOrder ? (
   689→              <div className="mt-14">
   690→                <div className="mb-6 text-center">
   691→                  <p className="font-display text-xl font-bold text-foreground md:text-2xl">
   692→                    অথবা কোর্সে <span className="text-gold-gradient">ভর্তি হতে চান?</span>

   695→                    Live classes, materials ও mock tests — ভর্তির পর সব Student Portal-এ।
   696→                  </p>

   698→                {hasHydrated && user && enrollments.length > 0 ? (
   699→                  <p className="mx-auto mb-6 flex max-w-xl items-center gap-2 rounded-xl border border-primary/20 bg-primary/5 px-4 py-3 text-sm text-primary">
   700→                    <BadgeCheck className="h-4 w-4 shrink-0" aria-hidden />
   701→                    আপনার পোর্টালে {enrollments.length}টি কোর্স আছে — নতুন কোর্স যোগ করতে পারেন।
   702→                  </p>
   703→                ) : null}
   704→                <CoursePicker />
   705→              </div>
   706→            ) : null}
   707→          </div>
   708→        </section>
   709→      </>
   710→    );
   711→  }
   712→
   713→  /* --- Already enrolled → portal CTA (but not during the success step) --- */
   714→  if (alreadyEnrolled && step !== 3) {
   715→    return (
   716→      <>
   717→        <PageHeader
   718→          eyebrow="Admission · Checkout"
   719→          title={

   736→              </p>
