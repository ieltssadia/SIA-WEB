   104→function CartSummary({ zone, onEdit }: { zone: string; onEdit?: () => void }) {

   106→  const setQuantity = useCartStore((s) => s.setQuantity);

   110→  const zoneInfo = getZone(zone);
   111→  const toFree = FREE_COURIER_THRESHOLD - subtotal;
   112→
   113→  return (
   114→    <div>

   118→        </p>
   119→        {onEdit ? (
   120→          <button
   121→            type="button"
   122→            onClick={onEdit}
   123→            className="text-[11px] font-semibold text-primary hover:underline"
   124→          >
   125→            Edit cart
   126→          </button>
   127→        ) : null}
   128→      </div>
   129→
   130→      <ul className="mt-4 space-y-3">
   131→        {items.map((item) => (
   132→          <li key={item.slug} className="flex gap-3">
   133→            <div className="relative h-16 w-12 shrink-0 overflow-hidden rounded-lg border border-primary/15">
   134→              <Image src={item.cover} alt="" fill sizes="48px" className="object-cover" />
   135→            </div>
   136→            <div className="min-w-0 flex-1">
   137→              <p className="line-clamp-2 text-xs font-semibold leading-snug text-foreground">
   138→                {item.title}
   139→              </p>
   140→              <div className="mt-1 flex items-center justify-between">
   141→                <div className="flex items-center rounded-md border border-border">
   142→                  <button
   143→                    type="button"
   144→                    aria-label={`Decrease quantity of ${item.title}`}
   145→                    onClick={() => setQuantity(item.slug, item.quantity - 1)}
   146→                    className="flex h-6 w-6 items-center justify-center text-muted-foreground transition-colors hover:text-primary"
   147→                  >
   148→                    <Minus className="h-3 w-3" aria-hidden />
   149→                  </button>
   150→                  <span className="w-6 text-center text-[11px] font-bold text-foreground">
   151→                    {item.quantity}
   152→                  </span>
   153→                  <button
   154→                    type="button"
   155→                    aria-label={`Increase quantity of ${item.title}`}
   156→                    onClick={() => setQuantity(item.slug, item.quantity + 1)}
   157→                    className="flex h-6 w-6 items-center justify-center text-muted-foreground transition-colors hover:text-primary"
   158→                  >
   159→                    <Plus className="h-3 w-3" aria-hidden />
   160→                  </button>
   161→                </div>
   162→                <span className="text-xs font-bold text-foreground">
   163→                  {taka(item.price * item.quantity)}
   164→                </span>
   165→              </div>
   166→            </div>
   167→          </li>
   168→        ))}
   169→      </ul>
   170→
   171→      <Separator className="my-4 bg-primary/10" />
   172→
   173→      <dl className="space-y-2 text-sm">
   174→        <div className="flex items-center justify-between">
   175→          <dt className="text-muted-foreground">Subtotal</dt>
   176→          <dd className="text-foreground">{taka(subtotal)}</dd>
   177→        </div>
   178→        <div className="flex items-center justify-between">
   179→          <dt className="text-muted-foreground">Delivery</dt>
   180→          <dd className={fee === 0 ? "font-medium text-emerald-400" : "text-foreground"}>
   181→            {fee === 0 ? "Free" : taka(fee)}
   182→          </dd>
   183→        </div>
   184→        <div className="flex items-center justify-between border-t border-primary/10 pt-2.5">
   185→          <dt className="font-display text-base font-bold text-foreground">Total</dt>
   186→          <dd className="font-display text-2xl font-bold text-brand-gradient">{taka(total)}</dd>
   187→        </div>
   188→      </dl>
   189→
   190→      {zoneInfo?.courier && toFree > 0 ? (
   191→        <p className="mt-3 flex items-start gap-1.5 text-[11px] leading-relaxed text-primary">
   192→          <Truck className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   193→          আর {taka(toFree)} যোগ করলেই কুরিয়ার ফ্রি ({taka(FREE_COURIER_THRESHOLD)}+ অর্ডারে)
   194→        </p>
   195→      ) : null}
   196→      {zoneInfo?.courier && toFree <= 0 ? (
   197→        <p className="mt-3 flex items-center gap-1.5 text-[11px] font-semibold text-emerald-400">
   198→          <Truck className="h-3.5 w-3.5 shrink-0" aria-hidden />
   199→          কুরিয়ার ডেলিভারি ফ্রি হয়ে গেছে!
   200→        </p>
   201→      ) : null}
   202→      {zoneInfo && !zoneInfo.courier ? (
   203→        <p className="mt-3 flex items-center gap-1.5 text-[11px] text-muted-foreground">
   204→          <MapPin className="h-3.5 w-3.5 shrink-0 text-primary" aria-hidden />
   205→          {zoneInfo.note}
   206→        </p>
   207→      ) : null}
   208→    </div>
   209→  );
   210→}

   668→
   669→export function CartCheckout() {
   670→  const items = useCartStore((s) => s.items);
   671→  const lastOrder = useCartStore((s) => s.lastOrder);
   672→  const setLastOrder = useCartStore((s) => s.setLastOrder);
   673→  const clear = useCartStore((s) => s.clear);
   674→  const hasHydrated = useCartStore((s) => s.hasHydrated);
   675→  const portalUser = usePortalStore((s) => s.user);
   676→
   677→  const [step, setStep] = useState<1 | 2 | 3>(1);
   678→  const [form, setForm] = useState<DetailsForm>(EMPTY_FORM);
   679→  const [formError, setFormError] = useState<string | null>(null);
   680→  const [method, setMethod] = useState<OrderPayMethod["id"]>("bKash");
   681→  const [busy, setBusy] = useState(false);
   682→
   683→  const subtotal = cartSubtotal(items);
   684→  const fee = deliveryFeeFor(form.zone, subtotal);
   685→  const total = subtotal + fee;
   686→
   687→  // Portal users get name/phone prefilled once the session restores.
   688→  if (hasHydrated && portalUser && form.name === "" && form.phone === "") {
   689→    setForm((f) => ({ ...f, name: portalUser.name, phone: portalUser.phone }));
   690→  }
   691→
   692→  function setField(k: keyof DetailsForm, v: string) {
   693→    setForm((f) => ({ ...f, [k]: v }));
   694→    setFormError(null);
   695→  }
   696→
   697→  function validateDetails(): string | null {
   698→    if (form.name.trim().length < 2) return "আপনার পুরো নাম লিখুন।";
   699→    const phone = canonicalPhone(form.phone);
   700→    if (!/^01[3-9]\d{8}$/.test(phone)) return "সঠিক মোবাইল নম্বর দিন — যেমন 01712-345678।";
   701→    if (form.email && !/^\S+@\S+\.\S+$/.test(form.email.trim())) return "Email টি সঠিক নয়।";
   702→    const zone = getZone(form.zone);
   703→    if (zone?.courier && form.address.trim().length < 8) {
   704→      return "কুরিয়ার ডেলিভারির জন্য সম্পূর্ণ ঠিকানা লিখুন (house/road/area)।";
   705→    }
   706→    return null;
   707→  }
   708→
   709→  function goNextFromDetails() {
   710→    const err = validateDetails();
   711→    if (err) {
   712→      setFormError(err);
   713→      return;
   714→    }
   715→    setFormError(null);
   716→    setStep(2);
   717→  }
   718→
   719→  async function placeOrder() {
   720→    setBusy(true);
   721→    setFormError(null);
   722→    try {
   723→      const res = await fetch("/api/orders", {
   724→        method: "POST",
   725→        headers: { "Content-Type": "application/json" },
   726→        body: JSON.stringify({
   727→          name: form.name.trim(),
   728→          phone: canonicalPhone(form.phone),
   729→          email: form.email.trim() || undefined,
   730→          zone: form.zone,
   731→          address: form.address.trim() || undefined,
   732→          note: form.note.trim() || undefined,
   733→          items: items.map((i) => ({ slug: i.slug, quantity: i.quantity })),
   734→          paymentMethod: method,
   735→          transactionId: form.transactionId.trim() || undefined,
   736→        }),
   737→      });
   738→      const data = (await res.json().catch(() => null)) as
   739→        | { order?: PlacedOrder; error?: string }
   740→        | null;
   741→      if (!res.ok || !data?.order) {
   742→        setFormError(data?.error ?? "Order placement failed — please try again.");
   743→        return;
   744→      }
   745→      clear();
   746→      setLastOrder(data.order);
   747→      setStep(3);
   748→      window.scrollTo({ top: 0, behavior: "smooth" });
   749→    } catch {
   750→      setFormError("Could not reach the server — check your connection.");
   751→    } finally {
   752→      setBusy(false);
   753→    }
   754→  }
   755→
   756→  /* --- Receipt view (just placed / restored from last session) --- */
   757→  if (hasHydrated && items.length === 0 && lastOrder) {
   758→    return (
   759→      <Reveal y={12}>
   760→        <div className="mx-auto max-w-3xl">
   761→          <OrderReceipt order={lastOrder} onNewOrder={() => setLastOrder(null)} />
   762→        </div>
   763→      </Reveal>
   764→    );
   765→  }
   766→
   767→  /* --- Empty cart --- */
   768→  if (!hasHydrated) {
   769→    return (
   770→      <div className="flex justify-center py-16">
   771→        <div className="h-8 w-8 animate-spin rounded-full border-2 border-primary/30 border-t-primary" aria-label="Loading cart" />
   772→      </div>
   773→    );
   774→  }
   775→
   776→  if (items.length === 0) {
   777→    return (
   778→      <div className="mx-auto max-w-xl">
   779→        <div className="rounded-3xl border border-border bg-card p-10 text-center">
   780→          <span className="mx-auto flex h-16 w-16 items-center justify-center rounded-full border border-primary/25 bg-primary/10">
   781→            <ShoppingBag className="h-7 w-7 text-primary/70" aria-hidden />
   782→          </span>
   783→          <h2 className="mt-4 font-display text-xl font-bold text-foreground">আপনার কার্ট খালি</h2>
   784→          <p className="mt-2 text-sm text-muted-foreground">
   785→            Book Shop থেকে বই কার্টে যোগ করুন — অথবা নিচের কোর্সে ভর্তি হোন।
   786→          </p>
   787→          <Button asChild className="mt-5 bg-brand-gradient px-6 py-6 font-semibold text-white hover:opacity-90">
   788→            <a href="#/shop">
   789→              <BookOpen className="mr-2 h-5 w-5" aria-hidden />
   790→              Browse Books
   791→            </a>
   792→          </Button>
   793→        </div>
   794→      </div>
   795→    );
   796→  }
   797→
   798→  /* --- Steps 1–2 flow --- */
   799→  return (
   800→    <div className="grid items-start gap-6 lg:grid-cols-[1fr_380px]">
   801→      <Reveal y={12}>
   802→        <Card className="border-primary/20 bg-card">
   803→          <CardContent className="space-y-6 p-6 md:p-8">
   804→            <Steps current={step} labels={["Details", "Payment", "Done"]} />
   805→
   806→            {step === 1 ? (
   807→              <DetailsStep form={form} setField={setField} onNext={goNextFromDetails} />
   808→            ) : null}
   809→
   810→            {step === 2 ? (
   811→              <PaymentStep
   812→                form={form}
   813→                setField={setField}
   814→                method={method}
   815→                setMethod={setMethod}
   816→                total={total}
   817→                onBack={() => setStep(1)}
   818→                onPlace={placeOrder}
   819→                busy={busy}
   820→                error={formError}
   821→              />
   822→            ) : null}
   823→
   824→            {formError && step === 1 ? <ErrorNote text={formError} /> : null}
   825→          </CardContent>
   826→        </Card>
   827→      </Reveal>
   828→
   829→      <Reveal y={12} delay={0.08}>
   830→        <Card className="border-primary/20 bg-card lg:sticky lg:top-24">
   831→          <CardContent className="p-6">
   832→            <CartSummary zone={form.zone} onEdit={() => setStep(1)} />
   833→            <Separator className="my-4 bg-primary/10" />
   834→            <ul className="space-y-2 text-xs text-muted-foreground">
   835→              <li className="flex items-start gap-2">
   836→                <BadgeCheck className="mt-0.5 h-3.5 w-3.5 shrink-0 text-primary" aria-hidden />
   837→                100% original — Sadia Rahman-এর লেখা বই

   839→              <li className="flex items-start gap-2">
   840→                <Truck className="mt-0.5 h-3.5 w-3.5 shrink-0 text-primary" aria-hidden />
