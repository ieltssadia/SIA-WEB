   337→   337→function PaymentStep({
   338→   338→  course,
   339→   339→  onEnrolled,
   340→   340→  onBack,
   341→   341→}: {
   342→   342→  course: Course;
   343→   343→  /** Called with the confirmed batch label once enrollment succeeds. */

   345→   345→  onBack: () => void;

   347→   347→  const token = usePortalStore((s) => s.token);
   348→   348→  const setSession = usePortalStore((s) => s.setSession);
   349→   349→  const logout = usePortalStore((s) => s.logout);
   350→   350→  const batchOptions = useMemo(() => nextBatchOptions(course.slug), [course.slug]);

   375→   375→          Authorization: `Bearer ${token ?? ""}`,
   376→   376→        },
   377→   377→        body: JSON.stringify({
   378→   378→          courseSlug: course.slug,
   379→   379→          batch,
   380→   380→          paymentMethod: method,
   381→   381→          transactionId: txId,
   382→   382→        }),
   383→   383→      });
   384→   384→      const data = (await res.json().catch(() => null)) as

   386→   386→            user?: { name: string; phone: string };
   387→   387→            enrollments?: never[];
   388→   388→            mocks?: never[];
   389→   389→            token?: string;
   390→   390→            error?: string;
   391→   391→          }
   392→   392→        | null;
   393→   393→      if (res.status === 401) {
   394→   394→        logout();
   395→   395→        setError("Session expired — please log in again.");
   396→   396→        onBack();
   397→   397→        return;
   398→   398→      }

   400→   400→        setError(data?.error ?? "Enrollment failed — please try again.");
   401→   401→        return;
   402→   402→      }
   403→   403→      setSession(data.user, data.enrollments ?? [], data.mocks ?? [], data.token ?? null);
   404→   404→      onEnrolled(batch);
   405→   405→    } catch {
   406→   406→      setError("Could not reach the server — check your connection.");

   410→   410→  }

   412→   412→  return (

   421→   421→              className={`flex cursor-pointer items-center gap-3 rounded-xl border p-3.5 transition-colors ${

   423→   423→              }`}
   424→   424→            >
   425→   425→              <RadioGroupItem id={`batch-${b.batch}`} value={b.batch} />
   426→   426→              <span className="min-w-0">
   427→   427→                <span className="block text-sm font-bold text-foreground">{b.batch}</span>
   428→   428→                <span className="block text-xs text-muted-foreground">{b.meta}</span>
   429→   429→              </span>

   443→   443→            return (
   444→   444→              <Label
   445→   445→                key={m.id}
   446→   446→                htmlFor={`pay-${m.id}`}
   447→   447→                className={`flex cursor-pointer items-center gap-3 rounded-xl border p-3.5 transition-colors ${

   449→   449→                }`}
   450→   450→              >
   451→   451→                <RadioGroupItem id={`pay-${m.id}`} value={m.id} />
   452→   452→                <span
