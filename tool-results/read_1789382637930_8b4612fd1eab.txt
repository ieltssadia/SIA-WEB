    28→const dhakaTime = new Intl.DateTimeFormat("en-GB", {
    29→  timeZone: "Asia/Dhaka",
    30→  hour: "2-digit",
    31→  minute: "2-digit",
    32→  hour12: true,

    35→function startLabel(iso: string): string {
    36→  return dhakaTime.format(new Date(iso));
    37→}
    38→
    39→/** "1h 05m 12s"-style elapsed clock. */
    40→function elapsedLabel(ms: number): string {
    41→  const total = Math.max(0, Math.floor(ms / 1000));
    42→  const h = Math.floor(total / 3600);
    43→  const m = Math.floor((total % 3600) / 60);
    44→  const s = total % 60;
    45→  const mm = String(m).padStart(2, "0");
    46→  const ss = String(s).padStart(2, "0");
    47→  return h > 0 ? `${h}:${mm}:${ss}` : `${m}:${ss}`;
    48→}
    49→
    50→/** Compact future countdown: "2d 3h", "4h 12m", "9m", "<1m". */
    51→function countdownLabel(ms: number): string {
    52→  if (ms <= 60_000) return "<1 মিনিট";
    53→  const mins = Math.floor(ms / 60_000);
    54→  if (mins < 60) return `${mins} মিনিট`;
    55→  const hours = Math.floor(mins / 60);
    56→  if (hours < 24) return `${hours} ঘণ্টা ${mins % 60} মিনিট`;
    57→  return `${Math.floor(hours / 24)} দিন ${hours % 24} ঘণ্টা`;
    58→}

   505→/* ── Stage ──────────────────────────────────────────────────────────── */
   506→
   507→function Stage({
   508→  slides,
   509→  slideIndex,
   510→  live,
   511→  startsAt,
   512→  role,
   513→  camOn,
   514→  initialFrame,
   515→  socketRef,
   516→  localStream,
   517→  now,
   518→}: {
   519→  slides: ClassSlide[];
   520→  slideIndex: number;
   521→  live: boolean;
   522→  startsAt: string;
   523→  role: Role;
   524→  camOn: boolean;
   525→  initialFrame: string | null;
   526→  socketRef: React.RefObject<import("socket.io-client").Socket | null>;
   527→  localStream: MediaStream | null;
   528→  now: number | null;
   529→}) {
   530→  const [frame, setFrame] = useState<string | null>(initialFrame);
   531→  const [reactions, setReactions] = useState<{ id: string; emoji: string; x: number }[]>([]);
   532→  const localVideoRef = useRef<HTMLVideoElement | null>(null);
   533→
   534→  // Live camera frames + floating reactions
   535→  useEffect(() => {
   536→    const socket = socketRef.current;
   537→    if (!socket) return;
   538→    const onFrame = (e: { data?: string }) => {
   539→      if (typeof e?.data === "string") setFrame(e.data);
   540→    };
   541→    const onCamStatus = (e: { on?: boolean }) => {
   542→      if (!e?.on) setFrame(null);
   543→    };
   544→    const onReact = (e: { emoji?: string }) => {
   545→      if (!e?.emoji) return;
   546→      const id = Math.random().toString(36).slice(2);
   547→      setReactions((prev) => [...prev.slice(-24), { id, emoji: e.emoji as string, x: 8 + Math.random() * 74 }]);
   548→      setTimeout(() => setReactions((prev) => prev.filter((r) => r.id !== id)), 2700);
   549→    };
   550→    socket.on("cam:frame", onFrame);
   551→    socket.on("cam:status", onCamStatus);
   552→    socket.on("reaction:new", onReact);
   553→    return () => {
   554→      socket.off("cam:frame", onFrame);
   555→      socket.off("cam:status", onCamStatus);
   556→      socket.off("reaction:new", onReact);
   557→    };
   558→  }, [socketRef]);
   559→
   560→  // Teacher's own local preview
   561→  useEffect(() => {
   562→    const v = localVideoRef.current;
   563→    if (v && localStream) {
   564→      v.srcObject = localStream;
   565→      v.play().catch(() => {});
   566→    }
   567→  }, [localStream]);
   568→
   569→  const slide = slides[Math.min(slideIndex, slides.length - 1)];
   570→  const showCam = camOn && (role === "teacher" ? !!localStream : !!frame);
   571→  const startsIn = now !== null ? new Date(startsAt).getTime() - now : null;
   572→  const isLate = startsIn !== null && startsIn <= 0;
   573→
   574→  return (
   575→    <div className="relative aspect-video w-full overflow-hidden bg-black" data-testid="classroom-stage">
   576→      {/* Main surface */}
   577→      {showCam ? (
   578→        role === "teacher" ? (
   579→          <video ref={localVideoRef} muted playsInline className="h-full w-full object-cover" />
   580→        ) : (
   581→          <img src={frame as string} alt="লাইভ ক্যামেরা" className="h-full w-full object-cover" />
   582→        )
   583→      ) : camOn ? (
   584→        <div className="flex h-full w-full items-center justify-center text-sm text-muted-foreground">
   585→          <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden />
   586→          ক্যামেরা সংযোগ হচ্ছে…
   587→        </div>
   588→      ) : (
   589→        <SlideView slide={slide} />
   590→      )}
   591→
   592→      {/* Slide PIP while the camera is live */}
   593→      {showCam ? (
   594→        <div className="absolute bottom-3 right-3 w-40 overflow-hidden rounded-lg border border-white/15 bg-[#0A241B]/95 shadow-xl sm:w-52">
   595→          <div className="border-b border-primary/20 px-2.5 py-1 text-[10px] font-semibold uppercase tracking-wider text-primary">
   596→            স্লাইড {slideIndex + 1}/{slides.length}
   597→          </div>
   598→          <p className="line-clamp-2 px-2.5 py-1.5 text-[11px] leading-snug text-foreground/90">{slide.title}</p>
   599→        </div>
   600→      ) : null}
   601→
   602→      {/* LIVE / waiting badge */}
   603→      {live ? (
   604→        <span className="absolute left-3 top-3 flex items-center gap-1.5 rounded-full bg-red-600 px-3 py-1 text-[11px] font-bold uppercase tracking-wider text-white shadow-lg">
   605→          <span className="relative flex h-2 w-2">
   606→            <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-white opacity-75" />
   607→            <span className="relative inline-flex h-2 w-2 rounded-full bg-white" />
   608→          </span>
   609→          LIVE
   610→        </span>
   611→      ) : (
   612→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-2 bg-black/72 px-6 text-center backdrop-blur-[2px]">
   613→          <span className="flex h-14 w-14 items-center justify-center rounded-full border border-primary/40 bg-primary/10">
   614→            <Radio className="h-6 w-6 text-primary" aria-hidden />
   615→          </span>
   616→          <p className="font-display text-lg font-bold text-foreground">ক্লাস এখনো লাইভ হয়নি</p>
   617→          <p className="text-sm text-muted-foreground">
   618→            {isLate
   619→              ? "ক্লাস শুরুর সময় হয়ে গেছে — শিক্ষক লাইভ করলেই এখানে দেখা যাবে।"
   620→              : startsIn !== null
   621→                ? `শুরু হতে বাকি ${countdownLabel(startsIn)} — শিক্ষক লাইভ করলেই এখানে দেখা যাবে।`
   622→                : "শিক্ষক লাইভ করলেই এখানে দেখা যাবে।"}
   623→          </p>
   624→          <p className="text-xs text-muted-foreground/70">
   625→            চ্যাটে প্রশ্ন করে রাখুন — Sadia Ma'am লাইভে উত্তর দেবেন।
   626→          </p>
   627→        </div>
   628→      )}
   629→
   630→      {/* Floating reactions */}
   631→      <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   632→        {reactions.map((r) => (
   633→          <span
   634→            key={r.id}
   635→            className="live-reaction absolute bottom-2 text-3xl drop-shadow-lg"
   636→            style={{ left: `${r.x}%` }}
   637→          >
   638→            {r.emoji}
   639→          </span>
   640→        ))}
   641→      </div>
   642→    </div>
   643→  );
   644→}

   677→/* ── Chat ───────────────────────────────────────────────────────────── */
   678→
   679→function ChatPanel({ chat, meId, onSend }: { chat: ChatMsg[]; meId: string | null; onSend: (text: string) => void }) {
   680→  const [text, setText] = useState("");
   681→  const scrollRef = useRef<HTMLDivElement | null>(null);
   682→
   683→  useEffect(() => {
   684→    const el = scrollRef.current;
   685→    if (el) el.scrollTop = el.scrollHeight;
   686→  }, [chat.length]);
   687→
   688→  return (
   689→    <div className="flex h-[380px] flex-col sm:h-[440px] lg:h-[560px]">
   690→      <div
   691→        ref={scrollRef}
   692→        className="live-scroll min-h-0 flex-1 space-y-2.5 overflow-y-auto px-4 py-3"
   693→        aria-label="ক্লাস চ্যাট"
   694→        aria-live="polite"
   695→      >
   696→        {chat.length === 0 ? (
   697→          <p className="pt-10 text-center text-xs text-muted-foreground">
   698→            ক্লাস শুরু হলেই এখানে কথা বলা যাবে — প্রথম মেসেজটি আপনিই পাঠান!
   699→          </p>
   700→        ) : null}
   701→        {chat.map((msg) =>
   702→          msg.role === "system" ? (
   703→            <p key={msg.id} className="text-center text-[11px] leading-relaxed text-muted-foreground/80">
   704→              {msg.text}
   705→            </p>
   706→          ) : (
   707→            <ChatBubble key={msg.id} msg={msg} own={!!msg.from && msg.from === meId} />
   708→          )
   709→        )}
   710→      </div>
   711→      <form
   712→        className="flex items-center gap-2 border-t border-primary/10 bg-[#0d0d11] p-3"
   713→        onSubmit={(e) => {
   714→          e.preventDefault();
   715→          const t = text.trim();
   716→          if (!t) return;
   717→          onSend(t);
   718→          setText("");
   719→        }}
   720→      >
   721→        <Input
   722→          value={text}
   723→          onChange={(e) => setText(e.target.value)}
   724→          placeholder="মেসেজ লিখুন…"
   725→          aria-label="চ্যাট মেসেজ"
   726→          maxLength={500}
   727→          className="h-10 border-primary/20 bg-[#16161b]"
   728→        />
   729→        <Button type="submit" size="icon" aria-label="পাঠান" className="h-10 w-10 shrink-0 bg-brand-gradient text-white hover:opacity-90">
   730→          <Send className="h-4 w-4" aria-hidden />
   731→        </Button>
   732→      </form>
   733→    </div>
   734→  );
   735→}
