   154→function AccountStep({ onAuthed }: { onAuthed: () => void }) {
   155→  const setSession = usePortalStore((s) => s.setSession);
   156→
   157→  const [mode, setMode] = useState<"signup" | "login">("signup");
   158→  const [name, setName] = useState("");
   159→  const [phone, setPhone] = useState("");
   160→  const [password, setPassword] = useState("");
   161→  const [confirm, setConfirm] = useState("");
   162→  const [error, setError] = useState<string | null>(null);
   163→  const [busy, setBusy] = useState(false);
   164→
   165→  async function submit(e: FormEvent<HTMLFormElement>) {
   166→    e.preventDefault();
   167→    setError(null);
   168→    if (mode === "signup" && password !== confirm) {
   169→      setError("Passwords don't match — আবার লিখুন।");
   170→      return;
   171→    }
   172→    setBusy(true);
   173→    try {
   174→      const res = await fetch(
   175→        mode === "signup" ? "/api/auth/register" : "/api/portal/login",
   176→        {
   177→          method: "POST",
   178→          headers: { "Content-Type": "application/json" },
   179→          body: JSON.stringify(
   180→            mode === "signup" ? { name, phone, password } : { phone, password }
   181→          ),
   182→        }
   183→      );
   184→      const data = (await res.json().catch(() => null)) as
   185→        | { user?: { name: string; phone: string }; token?: string; error?: string }
   186→        | null;
   187→      if (!res.ok || !data?.user) {
   188→        setError(data?.error ?? "Something went wrong — please try again.");
   189→        return;
   190→      }
   191→      setSession(data.user, [], [], data.token ?? null);
   192→      onAuthed();
   193→    } catch {
   194→      setError("Could not reach the server — check your connection.");
   195→    } finally {
   196→      setBusy(false);
   197→    }
   198→  }
   199→
   200→  return (
   201→    <div>
   202→      <h2 className="font-display text-lg font-bold text-foreground">
   203→        {mode === "signup" ? "Create your student account" : "Log in to continue"}
   204→      </h2>
   205→      <p className="mt-1 text-sm text-muted-foreground">
   206→        কোর্স কিনলে একই account-এ ভর্তি হবে — পোর্টালে সাথে সাথে দেখতে পাবেন।
   207→      </p>
   208→
   209→      <Tabs value={mode} onValueChange={(v) => { setMode(v as "signup" | "login"); setError(null); }} className="mt-4">
   210→        <TabsList className="grid w-full grid-cols-2 bg-[#101014]">
   211→          <TabsTrigger value="signup" className="gap-1.5 data-[state=active]:bg-gold-gradient data-[state=active]:text-[#16120a]">
   212→            <UserPlus className="h-4 w-4" aria-hidden /> Create account
   213→          </TabsTrigger>
   214→          <TabsTrigger value="login" className="gap-1.5 data-[state=active]:bg-gold-gradient data-[state=active]:text-[#16120a]">
   215→            <LogIn className="h-4 w-4" aria-hidden /> Log in
   216→          </TabsTrigger>
   217→        </TabsList>
   218→
   219→        <TabsContent value="signup">
   220→          <form onSubmit={submit} className="mt-4 space-y-3.5" noValidate>
   221→            <div className="space-y-1.5">
   222→              <Label htmlFor="co-name">Full name</Label>
   223→              <Input
   224→                id="co-name"
   225→                placeholder="e.g. Rahim Ahmed"
   226→                value={name}
   227→                onChange={(e) => { setName(e.target.value); setError(null); }}
   228→                autoComplete="name"
   229→              />
   230→            </div>
   231→            <PhoneField phone={phone} setPhone={setPhone} setError={setError} />
   232→            <div className="grid gap-3.5 sm:grid-cols-2">
   233→              <div className="space-y-1.5">
   234→                <Label htmlFor="co-pass">Password (min 6)</Label>
   235→                <Input
   236→                  id="co-pass"
   237→                  type="password"
   238→                  placeholder="Create a password"
   239→                  value={password}
   240→                  onChange={(e) => { setPassword(e.target.value); setError(null); }}
   241→                  autoComplete="new-password"
   242→                />
   243→              </div>
   244→              <div className="space-y-1.5">
   245→                <Label htmlFor="co-pass2">Confirm password</Label>
   246→                <Input
   247→                  id="co-pass2"
   248→                  type="password"
   249→                  placeholder="Repeat password"
   250→                  value={confirm}
   251→                  onChange={(e) => { setConfirm(e.target.value); setError(null); }}
   252→                  autoComplete="new-password"
   253→                />
   254→              </div>
   255→            </div>
   256→            {error ? <ErrorNote text={error} /> : null}
   257→            <Button type="submit" disabled={busy} className="w-full bg-gold-gradient font-semibold text-[#16120a] hover:opacity-90 disabled:opacity-60">
   258→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden /> : null}
   259→              Continue to payment
   260→              <ArrowRight className="ml-1.5 h-4 w-4" aria-hidden />
   261→            </Button>
   262→          </form>
   263→        </TabsContent>
   264→
   265→        <TabsContent value="login">
   266→          <form onSubmit={submit} className="mt-4 space-y-3.5" noValidate>
   267→            <PhoneField phone={phone} setPhone={setPhone} setError={setError} />
   268→            <div className="space-y-1.5">
   269→              <Label htmlFor="co-login-pass">Password</Label>
   270→              <Input
   271→                id="co-login-pass"
   272→                type="password"
   273→                placeholder="Your portal password"
   274→                value={password}
   275→                onChange={(e) => { setPassword(e.target.value); setError(null); }}
   276→                autoComplete="current-password"
   277→              />
   278→            </div>
   279→            {error ? <ErrorNote text={error} /> : null}
   280→            <Button type="submit" disabled={busy} className="w-full bg-gold-gradient font-semibold text-[#16120a] hover:opacity-90 disabled:opacity-60">
   281→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden /> : null}

   284→            </Button>
   285→          </form>
   286→        </TabsContent>
   287→      </Tabs>
   288→    </div>
   289→  );
   290→}

   336→
   337→function PaymentStep({
   338→  course,
   339→  onEnrolled,
   340→  onBack,
   341→}: {
   342→  course: Course;
   343→  /** Called with the confirmed batch label once enrollment succeeds. */
   344→  onEnrolled: (batch: string) => void;
   345→  onBack: () => void;
   346→}) {
   347→  const token = usePortalStore((s) => s.token);
   348→  const setSession = usePortalStore((s) => s.setSession);
   349→  const logout = usePortalStore((s) => s.logout);
   350→  const batchOptions = useMemo(() => nextBatchOptions(course.slug), [course.slug]);
   351→
   352→  const [batch, setBatch] = useState(batchOptions[0].batch);
   353→  const [method, setMethod] = useState<PayMethod["id"]>("bKash");
   354→  const [txId, setTxId] = useState("");
   355→  const [error, setError] = useState<string | null>(null);
   356→  const [busy, setBusy] = useState(false);
   357→
   358→  const selected = payMethods.find((m) => m.id === method)!;
   359→
   360→  // Reset batch if options change (course switched)
   361→  const [prevCourse, setPrevCourse] = useState(course.slug);
   362→  if (prevCourse !== course.slug) {
   363→    setPrevCourse(course.slug);
   364→    setBatch(batchOptions[0].batch);
   365→  }
   366→
   367→  async function confirm() {
   368→    setError(null);
   369→    setBusy(true);
   370→    try {
   371→      const res = await fetch("/api/portal/enroll", {
   372→        method: "POST",
   373→        headers: {
   374→          "Content-Type": "application/json",
   375→          Authorization: `Bearer ${token ?? ""}`,
   376→        },
   377→        body: JSON.stringify({
   378→          courseSlug: course.slug,
   379→          batch,
   380→          paymentMethod: method,
   381→          transactionId: txId,
   382→        }),
   383→      });
   384→      const data = (await res.json().catch(() => null)) as
   385→        | {
   386→            user?: { name: string; phone: string };
   387→            enrollments?: never[];
   388→            mocks?: never[];
   389→            token?: string;
   390→            error?: string;
   391→          }
   392→        | null;
   393→      if (res.status === 401) {
   394→        logout();
   395→        setError("Session expired — please log in again.");
   396→        onBack();
   397→        return;
   398→      }
   399→      if (!res.ok || !data?.user) {
   400→        setError(data?.error ?? "Enrollment failed — please try again.");
   401→        return;
   402→      }
   403→      setSession(data.user, data.enrollments ?? [], data.mocks ?? [], data.token ?? null);
   404→      onEnrolled(batch);
   405→    } catch {
   406→      setError("Could not reach the server — check your connection.");
   407→    } finally {
   408→      setBusy(false);
   409→    }
   410→  }
   411→
   412→  return (
   413→    <div className="space-y-6">
   414→      <div>
   415→        <h2 className="font-display text-lg font-bold text-foreground">Choose your batch</h2>
   416→        <RadioGroup value={batch} onValueChange={setBatch} className="mt-3 space-y-2">
   417→          {batchOptions.map((b) => (
   418→            <Label
   419→              key={b.batch}
   420→              htmlFor={`batch-${b.batch}`}
   421→              className={`flex cursor-pointer items-center gap-3 rounded-xl border p-3.5 transition-colors ${
   422→                batch === b.batch ? "border-primary/60 bg-primary/[0.07]" : "border-border bg-[#101014] hover:border-primary/30"
   423→              }`}
   424→            >
   425→              <RadioGroupItem id={`batch-${b.batch}`} value={b.batch} />
   426→              <span className="min-w-0">
   427→                <span className="block text-sm font-bold text-foreground">{b.batch}</span>
   428→                <span className="block text-xs text-muted-foreground">{b.meta}</span>
   429→              </span>
   430→              <CalendarDays className="ml-auto h-4 w-4 shrink-0 text-primary" aria-hidden />
   431→            </Label>
   432→          ))}
   433→        </RadioGroup>
   434→      </div>
   435→
   436→      <Separator className="bg-primary/10" />
   437→
   438→      <div>
   439→        <h2 className="font-display text-lg font-bold text-foreground">Payment method</h2>
   440→        <RadioGroup value={method} onValueChange={(v) => setMethod(v as PayMethod["id"])} className="mt-3 space-y-2">
   441→          {payMethods.map((m) => {
   442→            const Icon = m.icon;
   443→            return (
   444→              <Label
   445→                key={m.id}
   446→                htmlFor={`pay-${m.id}`}
   447→                className={`flex cursor-pointer items-center gap-3 rounded-xl border p-3.5 transition-colors ${
   448→                  method === m.id ? "border-primary/60 bg-primary/[0.07]" : "border-border bg-[#101014] hover:border-primary/30"
   449→                }`}
   450→              >
   451→                <RadioGroupItem id={`pay-${m.id}`} value={m.id} />
   452→                <span
   453→                  className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg text-white"
   454→                  style={{ backgroundColor: m.color }}
   455→                  aria-hidden
   456→                >
   457→                  <Icon className="h-4.5 w-4.5" />
   458→                </span>
   459→                <span className="min-w-0">
   460→                  <span className="block text-sm font-bold text-foreground">{m.label}</span>
   461→                  <span className="block truncate text-xs text-muted-foreground">{m.note}</span>
   462→                </span>
   463→              </Label>
   464→            );
   465→          })}
   466→        </RadioGroup>
   467→
   468→        {selected.needsTxId ? (
   469→          <div className="mt-3 space-y-1.5">
   470→            <Label htmlFor="co-txid">Transaction ID (optional here)</Label>
   471→            <Input
   472→              id="co-txid"
   473→              placeholder="Demo checkout — any text works"
   474→              value={txId}
   475→              onChange={(e) => setTxId(e.target.value)}
   476→              maxLength={60}
   477→            />
   478→            <p className="text-xs text-muted-foreground">
   479→              ডেমো মোড — আসল পেমেন্ট হয় না। প্রোডাকশনে TrxID ভেরিফিকেশন যুক্ত হবে।
   480→            </p>
   481→          </div>
   482→        ) : null}
   483→      </div>
   484→
   485→      {error ? <ErrorNote text={error} /> : null}
   486→
   487→      <div className="flex flex-col gap-2.5 sm:flex-row">
   488→        <Button
   489→          onClick={confirm}
   490→          disabled={busy}
   491→          className="flex-1 bg-gold-gradient py-6 text-base font-semibold text-[#16120a] shadow-[0_8px_30px_rgba(212,175,55,0.25)] hover:opacity-90 disabled:opacity-60"
   492→        >
   493→          {busy ? (
   494→            <>
   495→              <Loader2 className="mr-2 h-5 w-5 animate-spin" aria-hidden />
   496→              Processing...
   497→            </>
   498→          ) : (
   499→            <>
   500→              <ShieldCheck className="mr-2 h-5 w-5" aria-hidden />
   501→              Confirm Enrollment · {course.price === 0 ? "Free" : course.price ? taka(course.price) : "Custom"}
   502→            </>
   503→          )}
   504→        </Button>
   505→        <Button variant="outline" onClick={onBack} disabled={busy} className="border-primary/25 hover:text-primary">
   506→          Back
   507→        </Button>
   508→      </div>
   509→
   510→      <p className="flex items-center justify-center gap-1.5 text-xs text-muted-foreground">
   511→        <Lock className="h-3.5 w-3.5" aria-hidden />
   512→        Secure demo checkout · bKash / Nagad / Rocket / Bank / Cash accepted
   513→      </p>
   514→    </div>
   515→  );
   516→}

   610→/* Checkout page                                                       */
   611→/* ------------------------------------------------------------------ */
   612→
   613→export function CheckoutPage({ initialCourse }: { initialCourse: string | null }) {
   614→  const user = usePortalStore((s) => s.user);
   615→  const enrollments = usePortalStore((s) => s.enrollments);
   616→  const hasHydrated = usePortalStore((s) => s.hasHydrated);
   617→
   618→  const [courseSlug, setCourseSlug] = useState(initialCourse);
   619→  const [step, setStep] = useState<1 | 2 | 3>(1);
   620→  const [enrolledBatch, setEnrolledBatch] = useState<string | null>(null);
   621→
   622→  // Logged-in visitors skip account creation (render-time state adjustment).
   623→  // prevAuthed starts false so a visitor who arrives while ALREADY logged in
   624→  // still advances from step 1 → 2 on the first post-hydration render.
   625→  const authed = hasHydrated && !!user;
   626→  const [prevAuthed, setPrevAuthed] = useState(false);
   627→  if (authed !== prevAuthed) {
   628→    setPrevAuthed(authed);
   629→    if (authed) setStep((s) => (s === 1 ? 2 : s));
   630→    else setStep(1);
   631→  }
   632→
   633→  // The course comes from the hash query — sync prop → state when the
   634→  // visitor switches courses (render-time adjustment; useState ignores
   635→  // new prop values on re-render)
   636→  const [prevInitial, setPrevInitial] = useState(initialCourse);
   637→  if (initialCourse !== prevInitial) {
   638→    setPrevInitial(initialCourse);
   639→    setCourseSlug(initialCourse);
   640→    setEnrolledBatch(null);
   641→    setStep(authed ? 2 : 1);
   642→  }
   643→
   644→  const course = courses.find((c) => c.slug === courseSlug) ?? null;
   645→  const alreadyEnrolled = !!(course && user && enrollments.some((e) => e.courseSlug === course.slug));
   646→
   647→  /* --- No course chosen → picker --- */
   648→  if (!course) {
   649→    return (
   650→      <>
   651→        <PageHeader
   652→          eyebrow="Admission · Checkout"
   653→          title={
   654→            <>
   655→              Choose a <span className="text-gold-gradient">course</span>
   656→            </>
   657→          }
   658→          subtitle="যে কোর্সে ভর্তি হতে চান বেছে নিন — payment-এর পরই কোর্সটি আপনার Student Portal-এ চলে আসবে।"
   659→        />
   660→        <section className="py-10 md:py-14">
   661→          <div className="mx-auto max-w-7xl px-4 lg:px-8">
   662→            {hasHydrated && user && enrollments.length > 0 ? (
   663→              <p className="mb-6 flex items-center gap-2 rounded-xl border border-primary/20 bg-primary/5 px-4 py-3 text-sm text-primary">
   664→                <BadgeCheck className="h-4 w-4 shrink-0" aria-hidden />
   665→                আপনার পোর্টালে {enrollments.length}টি কোর্স আছে — নতুন কোর্স যোগ করতে পারেন।
   666→              </p>
   667→            ) : null}
   668→            <CoursePicker />
   669→          </div>
   670→        </section>
   671→      </>
   672→    );
   673→  }
   674→
   675→  /* --- Already enrolled → portal CTA (but not during the success step) --- */
   676→  if (alreadyEnrolled && step !== 3) {
   677→    return (
   678→      <>
   679→        <PageHeader
   680→          eyebrow="Admission · Checkout"
   681→          title={
   682→            <>
   683→              Already <span className="text-gold-gradient">enrolled</span>
   684→            </>
   685→          }
   686→        />
   687→        <section className="py-10 md:py-14">
   688→          <div className="mx-auto max-w-2xl px-4 lg:px-8">
   689→            <div className="rounded-3xl border border-primary/25 bg-card p-8 text-center">
   690→              <span className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-gold-gradient">
   691→                <BadgeCheck className="h-7 w-7 text-[#16120a]" aria-hidden />
   692→              </span>
   693→              <h2 className="mt-4 font-display text-xl font-bold text-foreground">
   694→                আপনি ইতিমধ্যে এই কোর্সে ভর্তি
   695→              </h2>
   696→              <p className="mt-2 text-sm text-muted-foreground">
   697→                {course.title} — আপনার পোর্টালে routine, materials ও scores দেখুন।
   698→              </p>
   699→              <Button asChild className="mt-5 bg-gold-gradient px-6 py-6 font-semibold text-[#16120a] hover:opacity-90">
   700→                <a href="#/portal">
   701→                  <GraduationCap className="mr-2 h-5 w-5" aria-hidden />
   702→                  Go to My Portal
   703→                </a>
   704→              </Button>
   705→            </div>
   706→          </div>
   707→        </section>
   708→      </>

   713→  return (

   722→        subtitle="২ ধাপে ভর্তি সম্পন্ন করুন — account, তারপর payment। কোর্স সাথে সাথেই পোর্টালে যুক্ত হবে।"
