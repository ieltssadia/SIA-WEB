    44→function formatBDT(n: number) {
    45→  return `৳${n.toLocaleString("en-US")}`;

    48→function discountPct(price: number, oldPrice?: number) {
    49→  if (!oldPrice || oldPrice <= price) return null;
    50→  return `-${Math.round((1 - price / oldPrice) * 100)}%`;
    51→}
    52→
    53→function formatStudents(n: number) {
    54→  return n >= 1000 ? `${(n / 1000).toFixed(1).replace(/\.0$/, "")}k+` : `${n}`;
