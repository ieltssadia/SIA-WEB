    52→    52→    52→function toMinutes(time: string): number {
    53→    53→    53→  const m = time.match(/(\d+):(\d+)\s*(AM|PM)/i);
    54→    54→    54→  if (!m) return 0;
    55→    55→    55→  let h = Number(m[1]) % 12;
    56→    56→    56→  if (/pm/i.test(m[3])) h += 12;
    57→    57→    57→  return h * 60 + Number(m[2]);
    58→    58→    58→}

    61→    61→    61→function findNextClass(now: Date): { row: RoutineClass; label: string } | null {

    65→    65→    65→
    66→    66→    66→  for (let offset = 0; offset < 8; offset++) {
    67→    67→    67→    const day = weekDays[(todayIdx + offset) % 7];
    68→    68→    68→    const rows = classRoutine
    69→    69→    69→      .filter((r) => r.day === day)
    70→    70→    70→      .sort((a, b) => toMinutes(a.start) - toMinutes(b.start));
    71→    71→    71→    for (const row of rows) {
    72→    72→    72→      if (offset === 0 && toMinutes(row.start) <= nowMinutes + 10) continue;

    74→    74→    74→      return { row, label };
