   158→function AccountStep({ onAuthed }: { onAuthed: () => void }) {
   159→  const setSession = usePortalStore((s) => s.setSession);
   160→
   161→  const [mode, setMode] = useState<"signup" | "login">("signup");
   162→  const [name, setName] = useState("");
   163→  const [phone, setPhone] = useState("");
   164→  const [password, setPassword] = useState("");
   165→  const [confirm, setConfirm] = useState("");
   166→  const [error, setError] = useState<string | null>(null);
   167→  const [busy, setBusy] = useState(false);
   168→
   169→  async function submit(e: FormEvent<HTMLFormElement>) {
   170→    e.preventDefault();
   171→    setError(null);
   172→    if (mode === "signup" && password !== confirm) {
   173→      setError("Passwords don't match — আবার লিখুন।");
   174→      return;
   175→    }
   176→    setBusy(true);
   177→    try {
   178→      const res = await fetch(
   179→        mode === "signup" ? "/api/auth/register" : "/api/portal/login",
   180→        {
   181→          method: "POST",
   182→          headers: { "Content-Type": "application/json" },
   183→          body: JSON.stringify(
   184→            mode === "signup" ? { name, phone, password } : { phone, password }
   185→          ),
   186→        }
   187→      );
   188→      const data = (await res.json().catch(() => null)) as
   189→        | { user?: { name: string; phone: string }; token?: string; error?: string }
   190→        | null;
   191→      if (!res.ok || !data?.user) {
   192→        setError(data?.error ?? "Something went wrong — please try again.");
   193→        return;
   194→      }
   195→      setSession(data.user, [], [], data.token ?? null);
   196→      onAuthed();
   197→    } catch {
   198→      setError("Could not reach the server — check your connection.");
   199→    } finally {
   200→      setBusy(false);
   201→    }
   202→  }
   203→
   204→  return (
   205→    <div>
   206→      <h2 className="font-display text-lg font-bold text-foreground">
   207→        {mode === "signup" ? "Create your student account" : "Log in to continue"}
   208→      </h2>
   209→      <p className="mt-1 text-sm text-muted-foreground">
   210→        কোর্স কিনলে একই account-এ ভর্তি হবে — পোর্টালে সাথে সাথে দেখতে পাবেন।
   211→      </p>
   212→
   213→      <Tabs value={mode} onValueChange={(v) => { setMode(v as "signup" | "login"); setError(null); }} className="mt-4">
   214→        <TabsList className="grid w-full grid-cols-2 bg-[#0A241B]">
   215→          <TabsTrigger value="signup" className="gap-1.5 data-[state=active]:bg-brand-gradient data-[state=active]:text-white">
   216→            <UserPlus className="h-4 w-4" aria-hidden /> Create account
   217→          </TabsTrigger>
   218→          <TabsTrigger value="login" className="gap-1.5 data-[state=active]:bg-brand-gradient data-[state=active]:text-white">
   219→            <LogIn className="h-4 w-4" aria-hidden /> Log in
   220→          </TabsTrigger>
   221→        </TabsList>
   222→
   223→        <TabsContent value="signup">
   224→          <form onSubmit={submit} className="mt-4 space-y-3.5" noValidate>
   225→            <div className="space-y-1.5">
   226→              <Label htmlFor="co-name">Full name</Label>
   227→              <Input
   228→                id="co-name"
   229→                placeholder="e.g. Rahim Ahmed"
   230→                value={name}
   231→                onChange={(e) => { setName(e.target.value); setError(null); }}
   232→                autoComplete="name"
   233→              />
   234→            </div>
   235→            <PhoneField phone={phone} setPhone={setPhone} setError={setError} />
   236→            <div className="grid gap-3.5 sm:grid-cols-2">
   237→              <div className="space-y-1.5">
   238→                <Label htmlFor="co-pass">Password (min 6)</Label>
   239→                <Input
   240→                  id="co-pass"
   241→                  type="password"
   242→                  placeholder="Create a password"
   243→                  value={password}
   244→                  onChange={(e) => { setPassword(e.target.value); setError(null); }}
   245→                  autoComplete="new-password"
   246→                />
   247→              </div>
   248→              <div className="space-y-1.5">
   249→                <Label htmlFor="co-pass2">Confirm password</Label>
   250→                <Input
   251→                  id="co-pass2"
   252→                  type="password"
   253→                  placeholder="Repeat password"
   254→                  value={confirm}
   255→                  onChange={(e) => { setConfirm(e.target.value); setError(null); }}
   256→                  autoComplete="new-password"
   257→                />
   258→              </div>
   259→            </div>
   260→            {error ? <ErrorNote text={error} /> : null}
   261→            <Button type="submit" disabled={busy} className="w-full bg-brand-gradient font-semibold text-white hover:opacity-90 disabled:opacity-60">
   262→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden /> : null}

   265→            </Button>
   266→          </form>
   267→        </TabsContent>
   268→
   269→        <TabsContent value="login">
   270→          <form onSubmit={submit} className="mt-4 space-y-3.5" noValidate>
   271→            <PhoneField phone={phone} setPhone={setPhone} setError={setError} />
   272→            <div className="space-y-1.5">
   273→              <Label htmlFor="co-login-pass">Password</Label>
   274→              <Input
   275→                id="co-login-pass"
   276→                type="password"
   277→                placeholder="Your portal password"
   278→                value={password}
   279→                onChange={(e) => { setPassword(e.target.value); setError(null); }}
   280→                autoComplete="current-password"
   281→              />
   282→            </div>
   283→            {error ? <ErrorNote text={error} /> : null}

   359→function PaymentStep({
   360→  course,
   361→  onEnrolled,
   362→  onBack,
   363→}: {
   364→  course: Course;
   365→  /** Called with the confirmed batch label once enrollment succeeds. */
   366→  onEnrolled: (batch: string) => void;
   367→  onBack: () => void;
   368→}) {
   369→  const token = usePortalStore((s) => s.token);
   370→  const setSession = usePortalStore((s) => s.setSession);
   371→  const logout = usePortalStore((s) => s.logout);
   372→  const batchOptions = useMemo(() => nextBatchOptions(course.slug), [course.slug]);
   373→
   374→  const [batch, setBatch] = useState(batchOptions[0].batch);
   375→  const [method, setMethod] = useState<PayMethod["id"]>("bKash");
   376→  const [txId, setTxId] = useState("");
   377→  const [error, setError] = useState<string | null>(null);
   378→  const [busy, setBusy] = useState(false);
   379→
   380→  const selected = payMethods.find((m) => m.id === method)!;
   381→
   382→  // Reset batch if options change (course switched)
   383→  const [prevCourse, setPrevCourse] = useState(course.slug);
   384→  if (prevCourse !== course.slug) {
   385→    setPrevCourse(course.slug);
   386→    setBatch(batchOptions[0].batch);
   387→  }
   388→
   389→  async function confirm() {
   390→    setError(null);
   391→    setBusy(true);
   392→    try {
   393→      const res = await fetch("/api/portal/enroll", {
   394→        method: "POST",
   395→        headers: {
   396→          "Content-Type": "application/json",
   397→          Authorization: `Bearer ${token ?? ""}`,
   398→        },
   399→        body: JSON.stringify({
   400→          courseSlug: course.slug,
   401→          batch,
   402→          paymentMethod: method,
   403→          transactionId: txId,
   404→        }),
   405→      });
   406→      const data = (await res.json().catch(() => null)) as
   407→        | {
   408→            user?: { name: string; phone: string };
   409→            enrollments?: never[];
   410→            mocks?: never[];
   411→            token?: string;
   412→            error?: string;
   413→          }
   414→        | null;
   415→      if (res.status === 401) {
   416→        logout();
   417→        setError("Session expired — please log in again.");
   418→        onBack();
   419→        return;
   420→      }
   421→      if (!res.ok || !data?.user) {
   422→        setError(data?.error ?? "Enrollment failed — please try again.");
   423→        return;
   424→      }
   425→      setSession(data.user, data.enrollments ?? [], data.mocks ?? [], data.token ?? null);
   426→      onEnrolled(batch);
   427→    } catch {
   428→      setError("Could not reach the server — check your connection.");
   429→    } finally {
   430→      setBusy(false);
   431→    }
   432→  }
   433→
   434→  return (
   435→    <div className="space-y-6">
   436→      <OfferCountdownBanner />
   437→      <div>
   438→        <h2 className="font-display text-lg font-bold text-foreground">Choose your batch</h2>
   439→        <RadioGroup value={batch} onValueChange={setBatch} className="mt-3 space-y-2">
   440→          {batchOptions.map((b) => (
   441→            <Label
   442→              key={b.batch}
   443→              htmlFor={`batch-${b.batch}`}
   444→              className={`flex cursor-pointer items-center gap-3 rounded-xl border p-3.5 transition-colors ${
   445→                batch === b.batch ? "border-primary/60 bg-primary/[0.07]" : "border-border bg-[#0A241B] hover:border-primary/30"
   446→              }`}
   447→            >
   448→              <RadioGroupItem id={`batch-${b.batch}`} value={b.batch} />
   449→              <span className="min-w-0">
   450→                <span className="block text-sm font-bold text-foreground">{b.batch}</span>
   451→                <span className="block text-xs text-muted-foreground">{b.meta}</span>
   452→              </span>
   453→              <CalendarDays className="ml-auto h-4 w-4 shrink-0 text-primary" aria-hidden />
   454→            </Label>
   455→          ))}
   456→        </RadioGroup>
   457→      </div>
   458→
   459→      <Separator className="bg-primary/10" />
   460→
   461→      <div>
   462→        <h2 className="font-display text-lg font-bold text-foreground">Payment method</h2>
   463→        <RadioGroup value={method} onValueChange={(v) => setMethod(v as PayMethod["id"])} className="mt-3 space-y-2">

   502→              ডেমো মোড — আসল পেমেন্ট হয় না। প্রোডাকশনে TrxID ভেরিফিকেশন যুক্ত হবে।
   503→            </p>
   504→          </div>

   506→      </div>

   508→      {error ? <ErrorNote text={error} /> : null}
   509→
   510→      <div className="flex flex-col gap-2.5 sm:flex-row">
   511→        <Button
   512→          onClick={confirm}
   513→          disabled={busy}
   514→          className="flex-1 bg-brand-gradient py-6 text-base font-semibold text-white shadow-[0_8px_30px_rgba(16,138,96,0.25)] hover:opacity-90 disabled:opacity-60"

   520→            </>
   521→          ) : (
   522→            <>
   523→              <ShieldCheck className="mr-2 h-5 w-5" aria-hidden />

   636→export function CheckoutPage({ initialCourse }: { initialCourse: string | null }) {
   637→  const user = usePortalStore((s) => s.user);
   638→  const enrollments = usePortalStore((s) => s.enrollments);
   639→  const hasHydrated = usePortalStore((s) => s.hasHydrated);
   640→  const cartItems = useCartStore((s) => s.items);
   641→  const cartLastOrder = useCartStore((s) => s.lastOrder);
   642→  const cartHydrated = useCartStore((s) => s.hasHydrated);
   643→
   644→  const [courseSlug, setCourseSlug] = useState(initialCourse);
   645→  const [step, setStep] = useState<1 | 2 | 3>(1);
   646→  const [enrolledBatch, setEnrolledBatch] = useState<string | null>(null);
   647→
   648→  // Logged-in visitors skip account creation (render-time state adjustment).
   649→  // prevAuthed starts false so a visitor who arrives while ALREADY logged in
   650→  // still advances from step 1 → 2 on the first post-hydration render.
   651→  const authed = hasHydrated && !!user;
   652→  const [prevAuthed, setPrevAuthed] = useState(false);
   653→  if (authed !== prevAuthed) {
   654→    setPrevAuthed(authed);
   655→    if (authed) setStep((s) => (s === 1 ? 2 : s));
   656→    else setStep(1);
   657→  }
   658→
   659→  // The course comes from the hash query — sync prop → state when the
   660→  // visitor switches courses (render-time adjustment; useState ignores
   661→  // new prop values on re-render)
   662→  const [prevInitial, setPrevInitial] = useState(initialCourse);
   663→  if (initialCourse !== prevInitial) {
   664→    setPrevInitial(initialCourse);
   665→    setCourseSlug(initialCourse);
   666→    setEnrolledBatch(null);
   667→    setStep(authed ? 2 : 1);
   668→  }
   669→
   670→  const course = courses.find((c) => c.slug === courseSlug) ?? null;
   671→  const alreadyEnrolled = !!(course && user && enrollments.some((e) => e.courseSlug === course.slug));
   672→
   673→  /* --- No course chosen → cart checkout (+ course picker when cart is empty) --- */
   674→  if (!course) {
   675→    const cartHasItems = cartHydrated && cartItems.length > 0;
   676→    const showReceipt = cartHydrated && !cartHasItems && !!cartLastOrder;
   677→    return (
   678→      <>
   679→        <PageHeader
   680→          eyebrow={cartHasItems || showReceipt ? "Book Shop · Checkout" : "Admission · Checkout"}
   681→          title={
   682→            cartHasItems ? (
   683→              <>
   684→                Review &amp; <span className="text-brand-gradient">place your order</span>

   697→            cartHasItems
   698→              ? "ডেলিভারি ডিটেইলস দিন, পেমেন্ট বেছে নিন — অর্ডার কনফার্ম করতে ২৪ ঘণ্টার মধ্যে আমরা কল দেব।"
   699→              : showReceipt

   702→          }
   703→        />
   704→        <section className="py-10 md:py-14">
   705→          <div className="mx-auto max-w-7xl px-4 lg:px-8">
   706→            <CartCheckout />
   707→
   708→            {/* Course picker stays reachable when the cart is empty */}
